	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = ec_exalted
	desc = ec_manage_exalted_culture_decision_desc
	selection_tooltip = ec_manage_exalted_culture_decision_tooltip
	confirm_text = ec_manage_exalted_culture_decision_confirm

	ai_check_interval = 0

		is_alive = yes

	}

	effect = {
		custom_tooltip = ec_manage_exalted_culture_decision_effect_tt
		trigger_event = ec_settings.1001
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

# AI-only — player reaches Inspire via Actions hub. Preserves prior AI pulse behavior.
ec_inspire_acceptance_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = ec_exalted
	desc = ec_inspire_acceptance_decision_desc
	selection_tooltip = ec_inspire_acceptance_decision_tooltip
	confirm_text = ec_inspire_acceptance_decision_confirm

	ai_check_interval = 36

	cooldown = { days = 365 }

	is_shown = {
		is_alive = yes
		is_ai = yes
		has_trait = ec_exalted_culturalist
		has_game_rule = ec_rule_acceptance_pulse_on
		NOT = { has_game_rule = ec_rule_granting_disabled }
	}

	is_valid = {
		custom_tooltip = {
			text = ec_tt_inspire_acceptance_ready
			is_landed = yes
			exists = culture
			NOT = { has_character_flag = ec_inspire_acceptance_cooldown }
		}
	}

	effect = {
		custom_tooltip = ec_inspire_acceptance_decision_effect_tt
		ec_inspire_acceptance_effect = yes
	}

	ai_potential = {
		has_game_rule = ec_rule_granting_player_and_ai
		has_trait = ec_exalted_culturalist
		ec_full_culture_package_active = yes
	}

	ai_will_do = {
		base = 20
		modifier = {
			add = 20
			ec_is_culture_head = yes
		}
	}
}
