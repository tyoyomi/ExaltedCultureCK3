# Exalted Culture — scripted triggers

ec_is_exalted_culturalist = {
	has_trait = ec_exalted_culturalist
}

# Soft peer: Supreme Sovereign trait key. Fails closed if SS is absent (may log once).
ec_character_is_supreme_sovereign = {
	has_trait = Supreme_Sovereign
}

ec_is_culture_head = {
	exists = culture
	exists = culture.culture_head
	this = culture.culture_head
}

ec_full_culture_package_active = {
	has_trait = ec_exalted_culturalist
	ec_is_culture_head = yes
}

ec_hybrid_assist_threshold_active = {
	has_game_rule = ec_rule_hybrid_assist_acceptance_and_threshold
}

ec_granting_disabled = {
	has_game_rule = ec_rule_granting_disabled
}

ec_actor_may_bestow_by_rule = {
	NOT = { has_game_rule = ec_rule_granting_disabled }
	trigger_if = {
		limit = { has_game_rule = ec_rule_granting_player_only }
		is_ai = no
	}
}

ec_patron_under_culturalist_cap = {
	# this = patron/actor. Binder list only (no full-world).
	trigger_if = {
		limit = { has_game_rule = ec_rule_max_culturalists_unlimited }
		always = yes
	}
	trigger_else = {
		custom_tooltip = {
			text = ec_tt_under_culturalist_cap
			NOT = {
				any_in_list = {
					variable = ec_bound_culturalists
					count >= ec_max_culturalists_value
					is_alive = yes
					has_trait = ec_exalted_culturalist
				}
			}
		}
	}
}

# this = holder; requires var:ec_patron
ec_in_patron_realm = {
	exists = var:ec_patron
	var:ec_patron = {
		is_alive = yes
		save_temporary_scope_as = ec_temp_patron
	}
	OR = {
		this = scope:ec_temp_patron
		is_courtier_of = scope:ec_temp_patron
		is_pool_guest_of = scope:ec_temp_patron
		is_foreign_court_or_pool_guest_of = scope:ec_temp_patron
		AND = {
			scope:ec_temp_patron = { is_landed = yes }
			any_liege_or_above = { this = scope:ec_temp_patron }
		}
		AND = {
			scope:ec_temp_patron = { is_landed = no }
			exists = host
			scope:ec_temp_patron = { exists = host }
			host = scope:ec_temp_patron.host
		}
	}
}

ec_can_actor_bestow = {
	is_alive = yes
	ec_actor_may_bestow_by_rule = yes
	ec_patron_under_culturalist_cap = yes
}

# Recipient eligible (recipient = this; scope:ec_patron or scope:actor = granter)
ec_can_receive_exalted_culturalist = {
	is_alive = yes
	NOT = { has_trait = ec_exalted_culturalist }
	trigger_if = {
		limit = {
			exists = scope:ec_patron
		}
		OR = {
			AND = {
				has_game_rule = ec_rule_self_grant_yes
				this = scope:ec_patron
			}
			is_courtier_of = scope:ec_patron
			is_pool_guest_of = scope:ec_patron
			is_foreign_court_or_pool_guest_of = scope:ec_patron
			AND = {
				scope:ec_patron = { is_landed = yes }
				any_liege_or_above = { this = scope:ec_patron }
			}
			AND = {
				scope:ec_patron = { is_landed = no }
				exists = host
				scope:ec_patron = { exists = host }
				host = scope:ec_patron.host
			}
		}
		trigger_if = {
			limit = { has_game_rule = ec_rule_self_grant_no }
			this != scope:ec_patron
		}
	}
	trigger_else_if = {
		limit = {
			exists = scope:actor
		}
		OR = {
			AND = {
				has_game_rule = ec_rule_self_grant_yes
				this = scope:actor
			}
			is_courtier_of = scope:actor
			is_pool_guest_of = scope:actor
			is_foreign_court_or_pool_guest_of = scope:actor
			AND = {
				scope:actor = { is_landed = yes }
				any_liege_or_above = { this = scope:actor }
			}
			AND = {
				scope:actor = { is_landed = no }
				exists = host
				scope:actor = { exists = host }
				host = scope:actor.host
			}
		}
		trigger_if = {
			limit = { has_game_rule = ec_rule_self_grant_no }
			this != scope:actor
		}
	}
	trigger_else = {
		always = no
	}
	trigger_if = {
		limit = { has_game_rule = ec_rule_require_culture_head_head_only }
		custom_tooltip = {
			text = ec_tt_must_be_culture_head
			ec_is_culture_head = yes
		}
	}
	trigger_if = {
		limit = { has_game_rule = ec_rule_mutual_exclusion_block_with_sovereign }
		custom_tooltip = {
			text = ec_tt_blocks_with_sovereign
			NOT = { ec_character_is_supreme_sovereign = yes }
		}
	}
}

ec_should_strip_exalted_culturalist = {
	has_trait = ec_exalted_culturalist
	has_game_rule = ec_rule_strip_on_leave_on
	NAND = {
		ec_in_patron_realm = yes
	}
}

ec_actor_is_patron_of_recipient = {
	scope:recipient = {
		has_trait = ec_exalted_culturalist
		exists = var:ec_patron
		var:ec_patron = scope:actor
	}
}
