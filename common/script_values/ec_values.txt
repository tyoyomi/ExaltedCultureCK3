ec_max_culturalists_value = {
	value = 1
	if = {
		limit = { has_game_rule = ec_rule_max_culturalists_1 }
		value = 1
	}
	else_if = {
		limit = { has_game_rule = ec_rule_max_culturalists_3 }
		value = 3
	}
	else_if = {
		limit = { has_game_rule = ec_rule_max_culturalists_5 }
		value = 5
	}
	else_if = {
		limit = { has_game_rule = ec_rule_max_culturalists_unlimited }
		value = 9999
	}
}

ec_exalted_culturalist_count_for_patron_value = {
	value = 0
	if = {
		limit = {
			any_in_list = {
				variable = ec_bound_culturalists
				count >= 1
			}
		}
		every_in_list = {
			variable = ec_bound_culturalists
			limit = {
				is_alive = yes
				has_trait = ec_exalted_culturalist
				exists = var:ec_patron
				var:ec_patron = root
			}
			add = 1
		}
	}
	else = {
		every_vassal = {
			limit = {
				has_trait = ec_exalted_culturalist
				exists = var:ec_patron
				var:ec_patron = root
			}
			add = 1
		}
		every_courtier_or_guest = {
			limit = {
				has_trait = ec_exalted_culturalist
				exists = var:ec_patron
				var:ec_patron = root
			}
			add = 1
		}
	}
}
